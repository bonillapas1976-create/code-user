CREATE DATABASE bdd_sena ;

  CREATE TABLE tb_ficha (
    id_ficha int(11) PRIMARY KEY, 
    fecha_inicio date,
    fecha_fin date,
    numero_ficha int ,
    programa varchar(30),
    estado varchar(11)
    );
    
  CREATE TABLE tb_instructores (
    id_instructor int(11)PRIMARY KEY,
    nombre varchar(55),
    tipo_doc varchar(30),
    No_de_doc int(11),
    especialidad varchar(45),
    correo varchar(25),
    telefono bigint(20)
    );
    
  CREATE TABLE tb_aprendices (
    id_aprendiz int(11)PRIMARY KEY,
    nombre_ap varchar(55),
    tipo_doc varchar(11),
    numero_doc int(23),
    telefono bigint(20) 
    );
    
  CREATE TABLE tb_competencia (
    id_competencia int(11)PRIMARY KEY,
    competencia varchar(12),
    cod_compe  varchar(25),
    descripcion varchar(44),
    duracion int(11) 
    );
    
    
    
  CREATE TABLE tb_administrativos (
    id_admin int(11)PRIMARY KEY,
    nombre_ad varchar(50),
    documento int(11),
    telefono_ad bigint(20),
    dependencia varchar(50)
    );
